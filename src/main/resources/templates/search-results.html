package com.busbooking.busbooking; import java.time.Duration; import java.time.LocalDate; import org.openqa.selenium.By; import org.openqa.selenium.WebDriver; import org.openqa.selenium.WebElement; import org.openqa.selenium.chrome.ChromeDriver; import org.openqa.selenium.support.ui.Select; import org.openqa.selenium.support.ui.WebDriverWait; import org.testng.Assert; import org.testng.annotations.AfterMethod; import org.testng.annotations.BeforeMethod; import org.testng.annotations.Test; public class BusSearchResultsTest { WebDriver driver; WebDriverWait wait; @BeforeMethod public void setUp() { driver = new ChromeDriver(); driver.manage().window().maximize(); wait = new WebDriverWait( driver, Duration.ofSeconds(10) ); } @Test public void verifyBusSearchResults() { // Step 1: Open homepage driver.get("http://localhost:8082"); // Step 2: Select source city Select source = new Select( wait.until(d -> d.findElement(By.id("source"))) ); source.selectByVisibleText("Hyderabad"); // Step 3: Select destination city Select destination = new Select( driver.findElement(By.id("destination")) ); destination.selectByVisibleText("Bengaluru"); // Step 4: Set a future journey date String journeyDate = LocalDate.now() .plusDays(7) .toString(); WebElement dateElement = driver.findElement(By.id("journeyDate")); dateElement.sendKeys(journeyDate); // If native date input entry is incorrect, set its value // using the standard yyyy-MM-dd format. if (!journeyDate.equals(dateElement.getAttribute("value"))) { ((org.openqa.selenium.JavascriptExecutor) driver) .executeScript( "arguments[0].value = arguments[1];", dateElement, journeyDate ); } Assert.assertEquals( dateElement.getAttribute("value"), journeyDate, "Journey date is incorrect" ); // Step 5: Click Search buses driver.findElement( By.cssSelector("button.search-button") ).click(); // Step 6: Wait for the search results route wait.until(d -> d.getCurrentUrl().contains("/search") ); // Step 7: Verify route and date WebElement route = wait.until( d -> d.findElement(By.cssSelector(".route")) ); String routeText = route.getText(); Assert.assertTrue( routeText.contains("Hyderabad"), "Source city is missing from results" ); Assert.assertTrue( routeText.contains("Bengaluru"), "Destination city is missing from results" ); Assert.assertTrue( routeText.contains(journeyDate), "Journey date is missing from results" ); // Step 8: Verify all three bus cards are displayed java.util.List busCards = driver.findElements(By.cssSelector(".bus-card")); Assert.assertEquals( busCards.size(), 3, "Expected exactly three bus cards" ); // Step 9: Verify bus names and fares verifyBus( "Express Travels", "₹750" ); verifyBus( "Royal Roadways", "₹900" ); verifyBus( "City Connect", "₹650" ); // Step 10: Verify seat-selection links exist java.util.List selectButtons = driver.findElements( By.cssSelector(".bus-card a.select-button") ); Assert.assertEquals( selectButtons.size(), 3, "Expected three Select Seats links" ); for (WebElement button : selectButtons) { Assert.assertTrue( button.isDisplayed(), "A Select Seats link is not visible" ); } System.out.println("=================================="); System.out.println("PASS: Bus search results verified!"); System.out.println("Route: " + routeText); System.out.println("Journey Date: " + journeyDate); System.out.println("Bus cards found: " + busCards.size()); System.out.println("Express Travels: ₹750"); System.out.println("Royal Roadways: ₹900"); System.out.println("City Connect: ₹650"); System.out.println("Select Seats links: " + selectButtons.size()); System.out.println("=================================="); } private void verifyBus(String busName, String expectedPrice) { WebElement busCard = driver.findElements( By.cssSelector(".bus-card") ).stream() .filter(card -> card.findElement( By.cssSelector(".bus-name") ).getText().equals(busName)) .findFirst() .orElseThrow(() -> new AssertionError( "Bus not found: " + busName )); String actualPrice = busCard.findElement( By.cssSelector(".price") ).getText(); Assert.assertEquals( actualPrice, expectedPrice, "Incorrect fare for " + busName ); Assert.assertTrue( busCard.findElement( By.cssSelector("a.select-button") ).getAttribute("href").contains("/seats"), "Select Seats link is incorrect for " + busName ); } @AfterMethod public void tearDown() { if (driver != null) { driver.quit(); } } }
